SELECT
    p.proname AS nama_fungsi,
    pg_catalog.pg_get_function_arguments(p.oid) AS argumen,
    pg_catalog.pg_get_function_result(p.oid) AS tipe_kembalian,
    p.prosrc AS kode_sumber
FROM pg_proc p
LEFT JOIN pg_namespace n ON n.oid = p.pronamespace
WHERE n.nspname = 'public' -- Hanya menampilkan fungsi di skema public (buatan Anda)
ORDER BY nama_fungsi;
