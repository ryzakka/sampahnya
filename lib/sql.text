-- ====================================================================
-- 1. PENAMBAHAN KOLOM PADA TABEL
-- ====================================================================

ALTER TABLE public.rooms
  ADD COLUMN IF NOT EXISTS world_event text,
  ADD COLUMN IF NOT EXISTS world_event_until timestamp with time zone,
  ADD COLUMN IF NOT EXISTS next_event_at timestamp with time zone DEFAULT (now() + interval '40 seconds');

ALTER TABLE public.player_state
  ADD COLUMN IF NOT EXISTS waste_level integer NOT NULL DEFAULT 0 CHECK (waste_level BETWEEN 0 AND 12),
  ADD COLUMN IF NOT EXISTS kost_tier integer NOT NULL DEFAULT 0 CHECK (kost_tier BETWEEN 0 AND 2),
  ADD COLUMN IF NOT EXISTS sabotage_cooldown_until timestamp with time zone,
  ADD COLUMN IF NOT EXISTS sealed_until timestamp with time zone;

-- ====================================================================
-- 2. FUNGSI PEMBANTU PERHITUNGAN & SINKRONISASI STATE RUANGAN
-- ====================================================================

CREATE OR REPLACE FUNCTION public.waste_capacity_for(p_level integer)
RETURNS numeric LANGUAGE sql IMMUTABLE AS $$
  SELECT CASE p_level
    WHEN 0 THEN 100
    WHEN 1 THEN 125
    WHEN 2 THEN 155
    WHEN 3 THEN 195
    WHEN 4 THEN 240
    WHEN 5 THEN 295
    WHEN 6 THEN 360
    WHEN 7 THEN 400
    WHEN 8 THEN 430
    WHEN 9 THEN 470
    WHEN 10 THEN 500
    WHEN 11 THEN 550
    WHEN 12 THEN 620
    ELSE 100
  END::numeric;
$$;

CREATE OR REPLACE FUNCTION public.waste_cost_for(p_next_level integer)
RETURNS numeric LANGUAGE sql IMMUTABLE AS $$
  SELECT CASE p_next_level
    WHEN 1 THEN 80
    WHEN 2 THEN 130
    WHEN 3 THEN 200
    WHEN 4 THEN 290
    WHEN 5 THEN 400
    WHEN 6 THEN 540
    WHEN 7 THEN 720
    WHEN 8 THEN 950
    WHEN 9 THEN 1250
    WHEN 10 THEN 1650
    WHEN 11 THEN 2150
    WHEN 12 THEN 2800
    ELSE 999999
  END::numeric;
$$;

CREATE OR REPLACE FUNCTION public.sync_room_state(p_room uuid)
RETURNS jsonb
LANGUAGE plpgsql
SECURITY DEFINER
AS $$
DECLARE
  v_room public.rooms%ROWTYPE;
  v_now timestamptz := now();
  v_dt numeric;
  v_pol_delta numeric := 0;
  v_p public.player_state%ROWTYPE;
  v_p_dt numeric;
  v_new_trash numeric;
  v_new_money numeric;
  v_cap numeric;
  v_ratio numeric;
  v_base_inc numeric;
  v_base_tr numeric;
  v_tier_inc_mult numeric;
  v_tier_tr_mult numeric;
  v_waste_tr_red numeric;
  v_waste_inc_bonus numeric;
  v_pol_mult numeric;
  v_trash_mult numeric;
  v_alive_count integer := 0;
  v_last_alive uuid;
  v_roll numeric;
  v_ev text;
BEGIN
  SELECT * INTO v_room FROM public.rooms WHERE id = p_room FOR UPDATE;
  IF NOT FOUND THEN
    RAISE EXCEPTION 'Ruangan tidak ditemukan.';
  END IF;

  IF v_room.status <> 'playing' THEN
    RETURN jsonb_build_object('server_now', v_now, 'status', v_room.status);
  END IF;

  -- Bersihkan world_event yang sudah selesai
  IF v_room.world_event IS NOT NULL AND v_room.world_event_until <= v_now THEN
    v_room.world_event := NULL;
    v_room.world_event_until := NULL;
  END IF;

  -- Picu world_event acak baru bila waktunya tiba
  IF v_room.world_event IS NULL AND v_room.next_event_at IS NOT NULL AND v_now >= v_room.next_event_at THEN
    v_roll := random();
    IF v_roll < 0.28 THEN
      v_ev := 'rain_flood';
    ELSIF v_roll < 0.55 THEN
      v_ev := 'new_semester';
    ELSIF v_roll < 0.78 THEN
      v_ev := 'hauler_strike';
    ELSE
      v_ev := 'eco_subsidy';
    END IF;
    v_room.world_event := v_ev;
    v_room.world_event_until := v_now + interval '22 seconds';
    v_room.next_event_at := v_now + interval '58 seconds';

    INSERT INTO public.game_events (room_id, type, amount)
    VALUES (p_room, 'event_' || v_ev, 22);
  END IF;

  -- Hitung perubahan polusi dari pemain yang memakai Insinerator Lv 8/9 atau Lv 11/12
  FOR v_p IN SELECT * FROM public.player_state WHERE room_id = p_room AND eliminated_at IS NULL LOOP
    IF v_p.waste_level = 8 THEN
      v_pol_delta := v_pol_delta + 0.25;
    ELSIF v_p.waste_level = 9 THEN
      v_pol_delta := v_pol_delta + 0.45;
    ELSIF v_p.waste_level = 11 THEN
      v_pol_delta := v_pol_delta - 0.15;
    ELSIF v_p.waste_level = 12 THEN
      v_pol_delta := v_pol_delta - 0.35;
    END IF;
  END LOOP;

  v_dt := GREATEST(0, EXTRACT(EPOCH FROM (v_now - v_room.pollution_updated_at)));
  -- Laju penurunan polusi alami -0.4/dtk (atau -0.8/dtk saat Subsidi Hijau)
  v_pol_delta := v_pol_delta - (CASE WHEN v_room.world_event = 'eco_subsidy' THEN 0.8 ELSE 0.4 END);
  v_room.pollution := LEAST(100, GREATEST(0, v_room.pollution + v_pol_delta * v_dt));
  v_room.pollution_updated_at := v_now;
  v_room.last_tick_at := v_now;

  -- Jika polusi menyentuh >= 90, picu Kabut Asap (Smog) selama 18 detik
  IF v_room.pollution >= 90 AND (v_room.smog_until IS NULL OR v_room.smog_until < v_now) THEN
    v_room.smog_until := v_now + interval '18 seconds';
  END IF;

  UPDATE public.rooms
  SET pollution = v_room.pollution,
      pollution_updated_at = v_room.pollution_updated_at,
      last_tick_at = v_room.last_tick_at,
      smog_until = v_room.smog_until,
      world_event = v_room.world_event,
      world_event_until = v_room.world_event_until,
      next_event_at = v_room.next_event_at
  WHERE id = p_room;

  -- Perbarui setiap pemain
  FOR v_p IN SELECT * FROM public.player_state WHERE room_id = p_room FOR UPDATE LOOP
    IF v_p.eliminated_at IS NOT NULL THEN
      CONTINUE;
    END IF;

    v_p_dt := GREATEST(0, EXTRACT(EPOCH FROM (v_now - v_p.updated_at)));
    v_cap := public.waste_capacity_for(v_p.waste_level);

    v_new_money := v_p.money + v_p.income_rate * v_p_dt;
    v_new_trash := v_p.trash + v_p.trash_rate * v_p_dt;

    -- Selesaikan angkut truk sampah bila waktunya sudah lewat
    IF v_p.hauler_amount > 0 AND (v_p.hauler_busy_until IS NULL OR v_p.hauler_busy_until <= v_now) THEN
      v_new_trash := GREATEST(0, v_new_trash - v_p.hauler_amount);
      v_p.hauler_amount := 0;
      v_p.hauler_busy_until := NULL;
    END IF;

    v_new_trash := GREATEST(0, v_new_trash);

    -- Cek eliminasi bila sampah penuh
    IF v_new_trash >= v_cap THEN
      v_new_trash := v_cap;
      v_p.eliminated_at := v_now;
      v_p.income_rate := 0;
      v_p.trash_rate := 0;
      INSERT INTO public.game_events (room_id, target_id, type)
      VALUES (p_room, v_p.player_id, 'eliminated');
    ELSE
      -- Hitung ulang income_rate & trash_rate berdasarkan kamar, kost_tier, waste_level, polusi, dan world_event
      v_tier_inc_mult := CASE v_p.kost_tier WHEN 1 THEN 1.6 WHEN 2 THEN 2.4 ELSE 1.0 END;
      v_tier_tr_mult  := CASE v_p.kost_tier WHEN 1 THEN 1.25 WHEN 2 THEN 1.5 ELSE 1.0 END;

      v_base_inc := (1.2 + v_p.rooms_count * 1.15) * v_tier_inc_mult;
      v_base_tr  := (0.12 + v_p.rooms_count * 0.085) * v_tier_tr_mult;

      v_waste_tr_red := CASE v_p.waste_level
        WHEN 0 THEN 0.00
        WHEN 1 THEN 0.02
        WHEN 2 THEN 0.04
        WHEN 3 THEN 0.07
        WHEN 4 THEN 0.10
        WHEN 5 THEN 0.14
        WHEN 6 THEN 0.18
        WHEN 7 THEN 0.26
        WHEN 8 THEN 0.42
        WHEN 9 THEN 0.62
        WHEN 10 THEN 0.55
        WHEN 11 THEN 0.75
        WHEN 12 THEN 1.05
        ELSE 0.00
      END;

      v_waste_inc_bonus := CASE v_p.waste_level
        WHEN 7 THEN 1.5
        WHEN 10 THEN 1.0
        WHEN 11 THEN 2.5
        WHEN 12 THEN 4.5
        ELSE 0.0
      END;

      v_ratio := v_new_trash / GREATEST(1, v_cap);
      v_trash_mult := CASE
        WHEN v_ratio >= 0.90 THEN 0.40
        WHEN v_ratio >= 0.75 THEN 0.60
        WHEN v_ratio >= 0.50 THEN 0.85
        ELSE 1.00
      END;
      -- Kost Sultan lebih sensitif terhadap bau sampah
      IF v_p.kost_tier = 2 AND v_ratio >= 0.50 THEN
        v_trash_mult := v_trash_mult * 0.75;
      END IF;

      v_pol_mult := CASE
        WHEN v_room.pollution >= 85 THEN 0.60
        WHEN v_room.pollution >= 60 THEN 0.75
        WHEN v_room.pollution >= 30 THEN 0.90
        ELSE 1.00
      END;

      v_p.income_rate := (v_base_inc * v_trash_mult * v_pol_mult) + v_waste_inc_bonus;

      -- Penalti bakar sampah beruntun
      IF v_p.burn_penalty_until IS NOT NULL AND v_p.burn_penalty_until > v_now THEN
        v_p.income_rate := v_p.income_rate * GREATEST(0.35, 1.0 - v_p.burn_stacks * 0.15);
      ELSE
        v_p.burn_stacks := 0;
      END IF;

      -- Penalti disegel akibat Sidak RT/Dinas
      IF v_p.sealed_until IS NOT NULL AND v_p.sealed_until > v_now THEN
        v_p.income_rate := v_p.income_rate * 0.5;
      END IF;

      -- Bonus Musim Mahasiswa Baru
      IF v_room.world_event = 'new_semester' THEN
        v_p.income_rate := v_p.income_rate * 1.35;
      END IF;

      -- Laju sampah bersih (bisa bernilai negatif bila Insinerator lebih kuat dari produksi sampah kamar)
      IF v_room.pollution >= 85 THEN
        v_base_tr := v_base_tr * 1.20;
      END IF;
      IF v_room.world_event = 'rain_flood' THEN
        v_base_tr := v_base_tr * 1.35;
      END IF;

      v_p.trash_rate := GREATEST(-0.80, v_base_tr - v_waste_tr_red);
    END IF;

    UPDATE public.player_state
    SET money = v_new_money,
        trash = v_new_trash,
        trash_capacity = v_cap,
        income_rate = ROUND(v_p.income_rate::numeric, 2),
        trash_rate = ROUND(v_p.trash_rate::numeric, 3),
        burn_stacks = v_p.burn_stacks,
        hauler_amount = v_p.hauler_amount,
        hauler_busy_until = v_p.hauler_busy_until,
        eliminated_at = v_p.eliminated_at,
        updated_at = v_now
    WHERE room_id = p_room AND player_id = v_p.player_id;

    IF v_p.eliminated_at IS NULL THEN
      v_alive_count := v_alive_count + 1;
      v_last_alive := v_p.player_id;
    END IF;
  END LOOP;

  -- Cek apakah permainan selesai (tersisa <= 1 pemain hidup)
  IF v_alive_count <= 1 THEN
    UPDATE public.rooms
    SET status = 'finished',
        finished_at = v_now,
        winner_id = v_last_alive
    WHERE id = p_room;
  END IF;

  RETURN jsonb_build_object('server_now', v_now, 'status', v_room.status);
END;
$$;

CREATE OR REPLACE FUNCTION public.game_tick(p_room uuid)
RETURNS jsonb LANGUAGE plpgsql SECURITY DEFINER AS $$
BEGIN
  RETURN public.sync_room_state(p_room);
END;
$$;

CREATE OR REPLACE FUNCTION public.start_game(p_room uuid)
RETURNS void LANGUAGE plpgsql SECURITY DEFINER AS $$
DECLARE
  v_room public.rooms%ROWTYPE;
  v_count integer;
  v_ready integer;
BEGIN
  SELECT * INTO v_room FROM public.rooms WHERE id = p_room FOR UPDATE;
  IF v_room.host_id <> auth.uid() THEN
    RAISE EXCEPTION 'Hanya Host yang dapat memulai permainan.';
  END IF;

  SELECT count(*), count(*) FILTER (WHERE is_ready)
  INTO v_count, v_ready
  FROM public.room_players WHERE room_id = p_room;

  IF v_count < 2 THEN
    RAISE EXCEPTION 'Minimal 2 pemain untuk memulai.';
  END IF;

  DELETE FROM public.player_state WHERE room_id = p_room;
  INSERT INTO public.player_state (
    room_id, player_id, money, trash, trash_capacity, rooms_count, defense,
    income_rate, trash_rate, waste_level, kost_tier
  )
  SELECT p_room, player_id, 120, 5, 100, 1, 0, 2.35, 0.20, 0, 0
  FROM public.room_players WHERE room_id = p_room;

  UPDATE public.rooms
  SET status = 'playing',
      started_at = now(),
      pollution = 0,
      pollution_updated_at = now(),
      smog_until = NULL,
      world_event = NULL,
      world_event_until = NULL,
      next_event_at = now() + interval '35 seconds'
  WHERE id = p_room;
END;
$$;

-- Bangun Kamar Baru
CREATE OR REPLACE FUNCTION public.build_room(p_room uuid)
RETURNS void LANGUAGE plpgsql SECURITY DEFINER AS $$
DECLARE
  v_room public.rooms%ROWTYPE;
  v_p public.player_state%ROWTYPE;
  v_cost numeric;
BEGIN
  PERFORM public.sync_room_state(p_room);
  SELECT * INTO v_room FROM public.rooms WHERE id = p_room;
  SELECT * INTO v_p FROM public.player_state WHERE room_id = p_room AND player_id = auth.uid() FOR UPDATE;

  IF v_p.eliminated_at IS NOT NULL THEN RAISE EXCEPTION 'Kost Anda sudah tertimbun.'; END IF;
  IF v_p.sealed_until IS NOT NULL AND v_p.sealed_until > now() THEN
    RAISE EXCEPTION 'Kost sedang disegel Dinas/RT! Tunggu segel dibuka.';
  END IF;
  IF v_p.rooms_count >= 12 THEN RAISE EXCEPTION 'Jumlah kamar sudah maksimal (12).'; END IF;

  v_cost := ROUND(
    100 * power(1.35, v_p.rooms_count)
    * (CASE WHEN v_room.pollution >= 60 THEN 1.5 ELSE 1.0 END)
    * (CASE WHEN v_room.world_event = 'new_semester' THEN 0.75 ELSE 1.0 END)
  );

  IF v_p.money < v_cost THEN RAISE EXCEPTION 'Uang belum cukup untuk membangun kamar.'; END IF;

  UPDATE public.player_state
  SET money = money - v_cost,
      rooms_count = rooms_count + 1
  WHERE room_id = p_room AND player_id = auth.uid();

  INSERT INTO public.game_events (room_id, actor_id, type, amount)
  VALUES (p_room, auth.uid(), 'build', v_p.rooms_count + 1);

  PERFORM public.sync_room_state(p_room);
END;
$$;

-- Upgrade Bank Sampah & Insinerator (Level 1..12)
CREATE OR REPLACE FUNCTION public.upgrade_waste(p_room uuid)
RETURNS void LANGUAGE plpgsql SECURITY DEFINER AS $$
DECLARE
  v_room public.rooms%ROWTYPE;
  v_p public.player_state%ROWTYPE;
  v_next integer;
  v_cost numeric;
BEGIN
  PERFORM public.sync_room_state(p_room);
  SELECT * INTO v_room FROM public.rooms WHERE id = p_room;
  SELECT * INTO v_p FROM public.player_state WHERE room_id = p_room AND player_id = auth.uid() FOR UPDATE;

  IF v_p.eliminated_at IS NOT NULL THEN RAISE EXCEPTION 'Kost Anda sudah tertimbun.'; END IF;
  IF v_p.waste_level >= 12 THEN RAISE EXCEPTION 'Fasilitas pengolahan limbah sudah level maksimal!'; END IF;

  v_next := v_p.waste_level + 1;
  v_cost := public.waste_cost_for(v_next);
  IF v_room.world_event = 'eco_subsidy' THEN
    v_cost := ROUND(v_cost * 0.70);
  END IF;

  IF v_p.money < v_cost THEN RAISE EXCEPTION 'Uang belum cukup untuk meningkatkan Bank Sampah.'; END IF;

  UPDATE public.player_state
  SET money = money - v_cost,
      waste_level = v_next,
      trash_capacity = public.waste_capacity_for(v_next)
  WHERE room_id = p_room AND player_id = auth.uid();

  INSERT INTO public.game_events (room_id, actor_id, type, amount)
  VALUES (p_room, auth.uid(), 'waste', v_next);

  PERFORM public.sync_room_state(p_room);
END;
$$;

-- Upgrade Kelas Kost (0 -> 1 Eksklusif -> 2 Sultan)
CREATE OR REPLACE FUNCTION public.upgrade_kost_tier(p_room uuid)
RETURNS void LANGUAGE plpgsql SECURITY DEFINER AS $$
DECLARE
  v_p public.player_state%ROWTYPE;
  v_next integer;
  v_cost numeric;
  v_min_rooms integer;
BEGIN
  PERFORM public.sync_room_state(p_room);
  SELECT * INTO v_p FROM public.player_state WHERE room_id = p_room AND player_id = auth.uid() FOR UPDATE;

  IF v_p.eliminated_at IS NOT NULL THEN RAISE EXCEPTION 'Kost Anda sudah tertimbun.'; END IF;
  IF v_p.kost_tier >= 2 THEN RAISE EXCEPTION 'Kelas kost sudah tingkat Sultan (Maksimal)!'; END IF;

  v_next := v_p.kost_tier + 1;
  v_cost := CASE v_next WHEN 1 THEN 650 ELSE 1600 END;
  v_min_rooms := CASE v_next WHEN 1 THEN 4 ELSE 8 END;

  IF v_p.rooms_count < v_min_rooms THEN
    RAISE EXCEPTION 'Butuh minimal % kamar untuk renovasi ke kelas ini.', v_min_rooms;
  END IF;
  IF v_p.money < v_cost THEN
    RAISE EXCEPTION 'Uang belum cukup untuk merenovasi kelas kost.';
  END IF;

  UPDATE public.player_state
  SET money = money - v_cost,
      kost_tier = v_next
  WHERE room_id = p_room AND player_id = auth.uid();

  INSERT INTO public.game_events (room_id, actor_id, type, amount)
  VALUES (p_room, auth.uid(), 'tier', v_next);

  PERFORM public.sync_room_state(p_room);
END;
$$;

-- Sewa Tukang Sampah
CREATE OR REPLACE FUNCTION public.hire_hauler(p_room uuid)
RETURNS void LANGUAGE plpgsql SECURITY DEFINER AS $$
DECLARE
  v_room public.rooms%ROWTYPE;
  v_p public.player_state%ROWTYPE;
  v_amt numeric;
  v_cost numeric;
BEGIN
  PERFORM public.sync_room_state(p_room);
  SELECT * INTO v_room FROM public.rooms WHERE id = p_room;
  SELECT * INTO v_p FROM public.player_state WHERE room_id = p_room AND player_id = auth.uid() FOR UPDATE;

  IF v_p.eliminated_at IS NOT NULL THEN RAISE EXCEPTION 'Kost Anda sudah tertimbun.'; END IF;
  IF v_room.world_event = 'hauler_strike' THEN
    RAISE EXCEPTION 'Tukang sampah sedang mogok kerja massal!';
  END IF;
  IF v_p.hauler_busy_until IS NOT NULL AND v_p.hauler_busy_until > now() THEN
    RAISE EXCEPTION 'Truk sampah sedang dalam perjalanan.';
  END IF;
  IF v_p.trash < 1 THEN RAISE EXCEPTION 'Belum ada sampah untuk diangkut.'; END IF;

  v_amt := LEAST(FLOOR(v_p.trash), 45 + v_p.waste_level * 4);
  v_cost := CEIL(v_amt * 3 * (CASE WHEN v_room.world_event = 'rain_flood' THEN 1.5 ELSE 1.0 END));

  IF v_p.money < v_cost THEN RAISE EXCEPTION 'Uang tidak cukup untuk menyewa truk sampah.'; END IF;

  UPDATE public.player_state
  SET money = money - v_cost,
      hauler_amount = v_amt,
      hauler_busy_until = now() + interval '7 seconds'
  WHERE room_id = p_room AND player_id = auth.uid();

  INSERT INTO public.game_events (room_id, actor_id, type, amount)
  VALUES (p_room, auth.uid(), 'haul', v_amt);
END;
$$;

-- Bakar Sampah Darurat
CREATE OR REPLACE FUNCTION public.burn(p_room uuid)
RETURNS void LANGUAGE plpgsql SECURITY DEFINER AS $$
DECLARE
  v_room public.rooms%ROWTYPE;
  v_p public.player_state%ROWTYPE;
  v_burn_amt numeric;
  v_pol_add numeric;
BEGIN
  PERFORM public.sync_room_state(p_room);
  SELECT * INTO v_room FROM public.rooms WHERE id = p_room FOR UPDATE;
  SELECT * INTO v_p FROM public.player_state WHERE room_id = p_room AND player_id = auth.uid() FOR UPDATE;

  IF v_p.eliminated_at IS NOT NULL THEN RAISE EXCEPTION 'Kost Anda sudah tertimbun.'; END IF;
  IF v_room.smog_until IS NOT NULL AND v_room.smog_until > now() THEN
    RAISE EXCEPTION 'Kabut asap pekat! Pembakaran sampah diblokir.';
  END IF;
  IF v_p.burn_cooldown_until IS NOT NULL AND v_p.burn_cooldown_until > now() THEN
    RAISE EXCEPTION 'Tungku pembakaran masih panas, tunggu sebentar.';
  END IF;
  IF v_p.trash < 2 THEN RAISE EXCEPTION 'Sampah terlalu sedikit untuk dibakar.'; END IF;

  v_burn_amt := LEAST(v_p.trash, 35 + v_p.waste_level * 3);
  -- Insinerator ramah lingkungan (Lv 10..12) menghasilkan polusi jauh lebih rendah saat bakar manual
  v_pol_add := CASE
    WHEN v_p.waste_level >= 10 THEN 6
    WHEN v_p.waste_level >= 8 THEN 22
    ELSE 15
  END;

  UPDATE public.player_state
  SET trash = GREATEST(0, trash - v_burn_amt),
      last_burn_at = now(),
      burn_cooldown_until = now() + interval '8 seconds',
      burn_penalty_until = CASE WHEN v_p.waste_level >= 10 THEN burn_penalty_until ELSE now() + interval '16 seconds' END,
      burn_stacks = CASE WHEN v_p.waste_level >= 10 THEN burn_stacks ELSE LEAST(4, burn_stacks + 1) END
  WHERE room_id = p_room AND player_id = auth.uid();

  UPDATE public.rooms
  SET pollution = LEAST(100, pollution + v_pol_add),
      smog_until = CASE WHEN pollution + v_pol_add >= 90 THEN now() + interval '18 seconds' ELSE smog_until END
  WHERE id = p_room;

  INSERT INTO public.game_events (room_id, actor_id, type, amount)
  VALUES (p_room, auth.uid(), 'burn', v_burn_amt);

  PERFORM public.sync_room_state(p_room);
END;
$$;

-- Beli Pertahanan
CREATE OR REPLACE FUNCTION public.buy_defense(p_room uuid)
RETURNS void LANGUAGE plpgsql SECURITY DEFINER AS $$
DECLARE
  v_p public.player_state%ROWTYPE;
  v_cost numeric;
BEGIN
  PERFORM public.sync_room_state(p_room);
  SELECT * INTO v_p FROM public.player_state WHERE room_id = p_room AND player_id = auth.uid() FOR UPDATE;

  IF v_p.eliminated_at IS NOT NULL THEN RAISE EXCEPTION 'Kost Anda sudah tertimbun.'; END IF;
  IF v_p.defense >= 3 THEN RAISE EXCEPTION 'Pertahanan sudah level maksimal.'; END IF;

  v_cost := 150 * (v_p.defense + 1);
  IF v_p.money < v_cost THEN RAISE EXCEPTION 'Uang belum cukup untuk memperkuat pertahanan.'; END IF;

  UPDATE public.player_state
  SET money = money - v_cost,
      defense = defense + 1,
      protected_until = now() + interval '10 seconds'
  WHERE room_id = p_room AND player_id = auth.uid();

  INSERT INTO public.game_events (room_id, actor_id, type, amount)
  VALUES (p_room, auth.uid(), 'defense', v_p.defense + 1);
END;
$$;

-- Sabotase 1: Lempar Sampah ke Lawan
CREATE OR REPLACE FUNCTION public.throw_trash(p_room uuid, p_target uuid)
RETURNS void LANGUAGE plpgsql SECURITY DEFINER AS $$
DECLARE
  v_me public.player_state%ROWTYPE;
  v_tg public.player_state%ROWTYPE;
  v_raw numeric;
  v_landed numeric;
BEGIN
  PERFORM public.sync_room_state(p_room);
  SELECT * INTO v_me FROM public.player_state WHERE room_id = p_room AND player_id = auth.uid() FOR UPDATE;
  SELECT * INTO v_tg FROM public.player_state WHERE room_id = p_room AND player_id = p_target FOR UPDATE;

  IF v_me.eliminated_at IS NOT NULL OR v_tg.eliminated_at IS NOT NULL THEN
    RAISE EXCEPTION 'Target tidak valid.';
  END IF;
  IF v_me.throw_cooldown_until IS NOT NULL AND v_me.throw_cooldown_until > now() THEN
    RAISE EXCEPTION 'Pelontar sampah sedang diisi ulang.';
  END IF;
  IF v_tg.protected_until IS NOT NULL AND v_tg.protected_until > now() THEN
    RAISE EXCEPTION 'Kost lawan sedang dilindungi perisai aktif!';
  END IF;
  IF v_me.money < 10 OR v_me.trash < 1 THEN
    RAISE EXCEPTION 'Butuh minimal 10 uang dan 1 sampah.';
  END IF;

  v_raw := LEAST(v_me.trash, 18 + v_me.rooms_count * 1.5);
  v_landed := ROUND(v_raw * (1.0 - v_tg.defense * 0.22));

  UPDATE public.player_state
  SET money = money - 10,
      trash = GREATEST(0, trash - v_raw),
      throw_cooldown_until = now() + interval '6 seconds'
  WHERE room_id = p_room AND player_id = auth.uid();

  UPDATE public.player_state
  SET trash = trash + v_landed,
      hit_streak = hit_streak + 1,
      last_hit_at = now(),
      protected_until = CASE WHEN hit_streak + 1 >= 3 THEN now() + interval '12 seconds' ELSE protected_until END
  WHERE room_id = p_room AND player_id = p_target;

  INSERT INTO public.game_events (room_id, actor_id, target_id, type, amount)
  VALUES (p_room, auth.uid(), p_target, 'throw', v_landed);

  PERFORM public.sync_room_state(p_room);
END;
$$;

-- Sabotase 2: Laporkan Sidak RT / Dinas Lingkungan
CREATE OR REPLACE FUNCTION public.sabotage_inspect(p_room uuid, p_target uuid)
RETURNS void LANGUAGE plpgsql SECURITY DEFINER AS $$
DECLARE
  v_me public.player_state%ROWTYPE;
  v_tg public.player_state%ROWTYPE;
  v_fine numeric;
  v_violating boolean;
BEGIN
  PERFORM public.sync_room_state(p_room);
  SELECT * INTO v_me FROM public.player_state WHERE room_id = p_room AND player_id = auth.uid() FOR UPDATE;
  SELECT * INTO v_tg FROM public.player_state WHERE room_id = p_room AND player_id = p_target FOR UPDATE;

  IF v_me.eliminated_at IS NOT NULL OR v_tg.eliminated_at IS NOT NULL THEN
    RAISE EXCEPTION 'Target tidak valid.';
  END IF;
  IF v_me.sabotage_cooldown_until IS NOT NULL AND v_me.sabotage_cooldown_until > now() THEN
    RAISE EXCEPTION 'Tim Sidak masih sibuk, tunggu jeda sabotase.';
  END IF;
  IF v_me.money < 120 THEN
    RAISE EXCEPTION 'Butuh 120 uang untuk menyewa Tim Sidak RT/Dinas.';
  END IF;

  v_violating := (v_tg.burn_penalty_until IS NOT NULL AND v_tg.burn_penalty_until > now())
                 OR (v_tg.waste_level IN (8, 9))
                 OR (v_tg.trash / GREATEST(1, v_tg.trash_capacity) >= 0.70);

  IF NOT v_violating THEN
    RAISE EXCEPTION 'Lawan sedang tertib! Sidak hanya bisa dikirim jika lawan habis bakar sampah, pakai Insinerator Berpolusi (Lv 8-9), atau sampahnya >70%%.';
  END IF;

  v_fine := ROUND(v_tg.money * 0.25);

  UPDATE public.player_state
  SET money = money - 120 + ROUND(v_fine * 0.6),
      sabotage_cooldown_until = now() + interval '22 seconds'
  WHERE room_id = p_room AND player_id = auth.uid();

  UPDATE public.player_state
  SET money = GREATEST(0, money - v_fine),
      sealed_until = now() + interval '14 seconds'
  WHERE room_id = p_room AND player_id = p_target;

  INSERT INTO public.game_events (room_id, actor_id, target_id, type, amount)
  VALUES (p_room, auth.uid(), p_target, 'inspect', v_fine);

  PERFORM public.sync_room_state(p_room);
END;
$$;

-- Sabotase 3: Bajak Truk Sampah Lawan
CREATE OR REPLACE FUNCTION public.sabotage_hijack(p_room uuid, p_target uuid)
RETURNS void LANGUAGE plpgsql SECURITY DEFINER AS $$
DECLARE
  v_me public.player_state%ROWTYPE;
  v_tg public.player_state%ROWTYPE;
  v_stolen numeric;
BEGIN
  PERFORM public.sync_room_state(p_room);
  SELECT * INTO v_me FROM public.player_state WHERE room_id = p_room AND player_id = auth.uid() FOR UPDATE;
  SELECT * INTO v_tg FROM public.player_state WHERE room_id = p_room AND player_id = p_target FOR UPDATE;

  IF v_me.eliminated_at IS NOT NULL OR v_tg.eliminated_at IS NOT NULL THEN
    RAISE EXCEPTION 'Target tidak valid.';
  END IF;
  IF v_me.sabotage_cooldown_until IS NOT NULL AND v_me.sabotage_cooldown_until > now() THEN
    RAISE EXCEPTION 'Jeda sabotase masih aktif.';
  END IF;
  IF v_tg.hauler_busy_until IS NULL OR v_tg.hauler_busy_until <= now() OR v_tg.hauler_amount <= 0 THEN
    RAISE EXCEPTION 'Lawan sedang tidak menyewa truk sampah!';
  END IF;
  IF v_me.money < 90 THEN
    RAISE EXCEPTION 'Butuh 90 uang untuk menyuap sopir truk lawan.';
  END IF;

  v_stolen := v_tg.hauler_amount;

  UPDATE public.player_state
  SET hauler_amount = 0,
      hauler_busy_until = NULL
  WHERE room_id = p_room AND player_id = p_target;

  UPDATE public.player_state
  SET money = money - 90,
      hauler_amount = v_stolen,
      hauler_busy_until = now() + interval '5 seconds',
      sabotage_cooldown_until = now() + interval '20 seconds'
  WHERE room_id = p_room AND player_id = auth.uid();

  INSERT INTO public.game_events (room_id, actor_id, target_id, type, amount)
  VALUES (p_room, auth.uid(), p_target, 'hijack', v_stolen);
END;
$$;
